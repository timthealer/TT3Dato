// src/archivist.ts
import { analyzeMessage } from './gemini';
import { callCEO, CEOSession } from './ceo';
import { Decision, ConversationState } from './types';
import { getProjects, getPeople, getAgents } from './registry';
import { createNote } from './notes';
import { normalizeText } from './alias';

export async function handleMessage(
  text: string,
  chatId: number,
  state: ConversationState
): Promise<any> {
  const step = state.step || 'idle';
  const data = state.data || {};

  switch (step) {
    case 'idle': {
      // Нормализуем текст (алиасы)
      const normalizedText = await normalizeText(text);
      const decision = await analyzeMessage(normalizedText);
      const projects = await getProjects();
      const people = await getPeople();
      const agents = await getAgents();

      const ceoDecision = await callCEO(decision, { projects, people, agents });

      if (ceoDecision.decision === 'ASK_USER') {
        // Сохраняем вопрос CEO в состоянии
        const session: CEOSession = {
          question: ceoDecision.message,
          decision
        };
        return {
          type: 'ask',
          message: ceoDecision.message,
          nextState: {
            step: 'waiting_ceo_answer',
            data: { session }
          }
        };
      }

      // Если решение принято — сохраняем
      let notePath = '';
      for (const action of ceoDecision.actions || []) {
        if (action.type === 'create_note') {
          notePath = await createNote(decision, ceoDecision);
        }
      }

      return {
        type: 'confirm',
        decision,
        ceoDecision,
        message: ceoDecision.message,
        notePath,
        nextState: { step: 'idle', data: {} }
      };
    }

    case 'waiting_ceo_answer': {
      const session = data.session as CEOSession;
      if (!session) {
        return {
          type: 'error',
          message: '❌ Ошибка: потерян контекст вопроса. Начните сначала.',
          nextState: { step: 'idle', data: {} }
        };
      }

      // Передаём ответ пользователя CEO
      const projects = await getProjects();

TT3Dato — что мы на самом деле строим
1. Главная идея

Изначально проект назывался Гек / TOS, затем ты переименовал его в:

TT3Dato — The Tolyan's Triple Dynamic Adaptive Technology Organism

При этом название для тебя имеет второй, скрытый смысл: TT → 3Dato → «пиздато». Это внутренняя шутка/пасхалка, которую наружу лучше не выносить.

Расшифровка TT3Dato задаёт и философию системы:

TT — The Tolyan
3 — Triple
D — Dynamic
A — Adaptive
T — Technology / Transformer
O — Organism

Три источника развития:

собственный код;
существующий Open Source;
обратная связь и данные пользователей.

То есть TT3Dato задуман не как очередной chatbot и не как один огромный AI-agent.

Это должна быть растущая система, которая со временем сама находит инструменты, анализирует их, предлагает изменения и адаптируется к новым задачам.

2. Каким ты хочешь видеть TT3Dato

В конечном счёте система должна уметь:

исследовать компании;
проводить аудит;
анализировать документы;
находить повторяющиеся процессы;
находить точки автоматизации;
искать существующие Open Source решения;
сравнивать их;
подключать лучшие;
создавать и изменять собственные компоненты только там, где готового решения нет;
работать с GitHub;
работать с Obsidian как с базой знаний;
использовать готовые agent skills;
создавать/подключать специализированных агентов;
координировать этих агентов;
накапливать знания;
постепенно перестраивать собственную архитектуру.

Но ключевой принцип:

TT3Dato не должен превращаться в бесконечную цепочку "агент управляет агентом, который управляет агентом".

Мы специально пришли к выводу, что сначала нужен маленький работающий MVP, а не огромная multi-agent система.

Мы отказались от подхода:

«Давайте сами напишем всех агентов».

Open Source First.

Если существует хороший готовый компонент — сначала изучить его.

Если существует несколько:

найти минимум несколько альтернатив;
сравнить;
оценить зрелость;
активность;
GitHub;
stars;
commits;
архитектуру;
лицензирование;
интеграцию;
выбрать лучший.

И только если подходящего решения нет — писать своё.

Это стало одним из главных принципов TT3Dato.
